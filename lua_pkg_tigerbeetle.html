TigerBeetle
Lua API

Справочник функций, доступных из Lua: параметры, возвращаемые значения, коды ошибок и готовые примеры с выводом через стандартный print.

Инициализация модуля

local pkg = require("pkg/tigerbeetle")

Важные типы

ID передаются как HEX-строки Uint128. amount передается как десятичная строка. Это позволяет не терять точность больших чисел в Lua.

Коды ошибок

КодТипОписание
0OKОперация выполнена успешно.
1Invalid IDНекорректный TigerBeetle Uint128 ID.
2Invalid AmountНекорректная сумма transfer.
3Invalid ParameterНекорректный параметр, тип, диапазон или batch-массив.
100ClientОшибка инициализации/работы TigerBeetle client.
101TigerBeetleTigerBeetle вернул status, отличный от Created, или неожиданный ответ.
404Not FoundAccount или transfer не найден.
Ничего не найдено. Измените строку поиска.

ID

Генерирует новый уникальный TigerBeetle Uint128 ID и возвращает его как HEX-строку.

#

Вызов

local id, err, code = pkg.ID()

Входящие параметры

Параметров нет.

Результат

РезультатТипОписание
idstringНовый Uint128 ID в HEX-представлении.
errstringПустая строка "" при успехе.
codeinteger0 при успехе.
• ID возвращается как HEX-строка. Для хранения ID в Lua лучше всегда использовать string.

Примеры

Создать один ID
local pkg = require("pkg/tigerbeetle")

local id, err, code = pkg.ID()

print("err =", err)
print("code =", code)

code = 0

local pkg = require("pkg/tigerbeetle")

local accountId1, err1, code1 = pkg.ID()

print("accountId1 =", accountId1)
print("err1 =", err1)
print("code1 =", code1)

print("accountId2 =", accountId2)

Ping

Проверяет, что клиент TigerBeetle может быть инициализирован с текущими настройками подключения.

#

Вызов

local ok, err, code = pkg.Ping()

Входящие параметры

Параметров нет.

Результат

РезультатТипОписание
okbooleantrue при успешной инициализации клиента, false при ошибке.
errstringПустая строка "" при успехе или текст ошибки.
codeinteger0 при успехе, 100 при ошибке клиента.
• Функция проверяет инициализацию клиента. Она не создает счета и переводы.

Примеры

Обычная проверка
local pkg = require("pkg/tigerbeetle")

err =

local pkg = require("pkg/tigerbeetle")

local ok, err, code = pkg.Ping()

if not ok then

    print("code =", code)
else
    print("TigerBeetle доступен")
    print("ok =", ok)
    print("err =", err)
    print("code =", code)

CreateAccount

Создает один простой счет TigerBeetle без дополнительных user data и flags.

#

Вызов

local result, err, code = pkg.CreateAccount(id, ledger, accountCode)

Входящие параметры

ПараметрТипОписание
idstringОбязательный Uint128 ID счета в HEX-представлении.
ledgerintegerОбязательный номер ledger, диапазон uint32.
accountCodeintegerОбязательный код типа счета, диапазон 0..65535.

Результат

РезультатТипОписание
resulttable|nilТаблица { id, status, timestamp } или nil при ошибке до получения результата.
errstringПустая строка "" при успехе или текст ошибки/status.
codeinteger0 при AccountCreated; 1/3/100/101 при ошибках.
• flags = 0, user_data_* = 0, timestamp задается TigerBeetle.

Примеры

Создать счет
local pkg = require("pkg/tigerbeetle")

local id, idErr, idCode = pkg.ID()
print("id =", id)
print("idErr =", idErr)
print("idCode =", idCode)

local result, err, code = pkg.CreateAccount(
    id,
    1,
    718
)

print("result.id =", result and result.id)
print("result.status =", result and result.status)
print("result.timestamp =", result and result.timestamp)
print("err =", err)
print("code =", code)
Пример вывода
result.id = 019d2b8c...
result.status = AccountCreated
result.timestamp = 178804...
err =
code = 0
Проверить ошибку создания
local pkg = require("pkg/tigerbeetle")

local id, _, _ = pkg.ID()

local result, err, code = pkg.CreateAccount(id, 1, 718)

if code ~= 0 then
    print("Счет не создан")
    print("result =", result)
    print("err =", err)
    print("code =", code)
else
    print("Счет создан")
    print("id =", result.id)
    print("status =", result.status)
    print("timestamp =", result.timestamp)
    print("err =", err)
    print("code =", code)
end

Счет создан

CreateAccounts

Пакетно создает несколько счетов одним вызовом TigerBeetle CreateAccounts.

#

Вызов

local results, err, code = pkg.CreateAccounts(accounts)

Входящие параметры

ПараметрТипОписание
accountstable[]Последовательный Lua-массив с индексами 1..N.
accounts[i].idstringОбязательно. Uint128 ID счета в HEX.
accounts[i].ledgerintegerОбязательно. uint32.
accounts[i].codeintegerОбязательно. uint16.
accounts[i].flagsintegerНеобязательно. uint16, по умолчанию 0.
accounts[i].user_data_128stringНеобязательно. HEX, по умолчанию "0".
accounts[i].user_data_64string|integerНеобязательно. uint64, по умолчанию 0.
accounts[i].user_data_32integerНеобязательно. uint32, по умолчанию 0.

Результат

РезультатТипОписание
resultstable[]|nilДля каждого элемента: { index, id, status, timestamp }.
errstring"" если все счета созданы; иначе первая обнаруженная ошибка/status.
codeinteger0 если все AccountCreated; 101 при TigerBeetle status error; другие коды при валидации/клиенте.
• Массив должен иметь последовательные индексы 1..N без пропусков.
• Если один status не AccountCreated, results все равно содержит ответы TigerBeetle для пакета.

Примеры

Создать два простых счета
local pkg = require("pkg/tigerbeetle")

local id1, _, _ = pkg.ID()
local id2, _, _ = pkg.ID()

local results, err, code = pkg.CreateAccounts({
    {
        id = id1,
        ledger = 1,
        code = 718
    },
    {
        id = id2,
        ledger = 1,
        code = 718
    }
})

print("err =", err)
print("code =", code)

if results then
    for i, r in ipairs(results) do
        print("result", i)
        print("index =", r.index)
        print("id =", r.id)
        print("status =", r.status)
        print("timestamp =", r.timestamp)
    end
end
Пример вывода

code = 0
result 1
index = 1
id = 019d...
status = AccountCreated
timestamp = 178804...
result 2
index = 2
id = 019d...
status = AccountCreated
timestamp = 178804...
Пакет с flags и user data
local pkg = require("pkg/tigerbeetle")

local id1, _, _ = pkg.ID()
local id2, _, _ = pkg.ID()

local historyFlags = pkg.AccountFlags(0, 0, 0, 1)
print("historyFlags =", historyFlags)

local results, err, code = pkg.CreateAccounts({
    {
        id = id1,

        id = id2,

print("results =", results)
print("err =", err)

    for _, r in ipairs(results) do
        print(r.index, r.id, r.status, r.timestamp)
    end
end

#

ПараметрТипОписание
idstringUint128 ID счета в HEX.

Результат

РезультатТипОписание
accounttable|nilПолная таблица счета или nil, если счет не найден/ошибка.
errstring"" при успехе; "account not found" при отсутствии счета.

ПараметрТипОписание

transfers[i].amountstring|integerОбязательно. Десятичная сумма.
transfers[i].ledgerintegerОбязательно. uint32.
transfers[i].codeintegerОбязательно. uint16.
transfers[i].flagsintegerНеобязательно. uint16, default 0.
transfers[i].pending_idstringНеобязательно. HEX или "0".
transfers[i].timeoutintegerНеобязательно. uint32, default 0.

        debit_account_id = accountA,
        credit_account_id = accountB,
        amount = "100",
        ledger = 1,
        code = 100
    },
    {
        id = transferId2,
